Usage

<!-- Basic switch -->
<app-switch 
  [checked]="isEnabled" 
  (checkedChange)="onSwitchChange($event)">
</app-switch>

<!-- Disabled switch -->
<app-switch 
  [checked]="isEnabled" 
  [disabled]="true">
</app-switch>

<!-- Two-way binding -->
<app-switch [(checked)]="isEnabled"></app-switch>

Properties

Property Type Default Description
checked boolean false Whether the switch is checked/on
disabled boolean false Whether the switch is disabled

Events

Event Payload Description
checkedChange boolean Emitted when the switch state changes
